# Личный номер инженера — вектор формата ENG-010.
#
# Номер, который инженер называет устройству и который сервер ищет в реестре.
# Читают этот файл обе стороны канала: крейт контракта и реализация на C#.
# Расхождение здесь означает, что парк напечатает номера, которые одна сторона
# принимает, а другая отвергает.
#
# Как читается номер: сегмент организации — ведущий ряд БУКВ, серийная часть —
# цифры, последний знак — контрольный символ ISO 7064 MOD 37,36 над всем, что
# перед ним. Отсюда и разбор `ORG1-000001`: сегмент `ORG`, серийная часть
# `1000001`. Цифра в `ORG1` написана для человека и считается серийной.
#
# Разделители арифметика игнорирует: три написания ниже — один номер.

body = ORG1-000001
number = ORG1-0000014
significant = ORG10000014
organisation_segment = ORG
serial = 1000001
check_character = 4

# Те же значащие символы, записанные иначе. Разбор обязан дать тот же номер.
spelling_lowercase = org1-0000014
spelling_spaces = ORG1 000001 4
spelling_dots = ORG1.000001.4

# Номер с испорченным контрольным символом: отвергается обеими сторонами.
